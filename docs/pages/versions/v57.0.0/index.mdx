import { useState, useEffect } from 'react';
import {
  StyleSheet,
  Text,
  View,
  TouchableOpacity,
  TextInput,
  ScrollView,
} from 'react-native';

const benches = [
  { id: 1, name: 'Ławka SIADAJ #001', free: 8, x: 18, y: 20 },
  { id: 2, name: 'Ławka SIADAJ #002', free: 5, x: 65, y: 18 },
  { id: 3, name: 'Ławka SIADAJ #003', free: 2, x: 35, y: 42 },
  { id: 4, name: 'Ławka SIADAJ #004', free: 7, x: 78, y: 50 },
  { id: 5, name: 'Ławka SIADAJ #005', free: 3, x: 15, y: 68 },
  { id: 6, name: 'Ławka SIADAJ #006', free: 10, x: 62, y: 75 },
];

const helpCategories = [
  {
    id: 'animals',
    name: 'Schroniska',
    emoji: '🐕',
    percentage: 47.8,
  },
  {
    id: 'hospitals',
    name: 'Szpitale',
    emoji: '🏥',
    percentage: 28.4,
  },
  {
    id: 'children',
    name: 'Domy dziecka',
    emoji: '🧸',
    percentage: 15.7,
  },
  {
    id: 'seniors',
    name: 'Domy seniora',
    emoji: '👵',
    percentage: 8.1,
  },
];

const previousHelp = [
  {
    month: 'Wrzesień 2026',
    cause: 'Schroniska',
    emoji: '🐕',
    amount: '290 575 zł',
  },
  {
    month: 'Sierpień 2026',
    cause: 'Szpitale',
    emoji: '🏥',
    amount: '184 320 zł',
  },
  {
    month: 'Lipiec 2026',
    cause: 'Domy dziecka',
    emoji: '🧸',
    amount: '126 840 zł',
  },
];

function getBenchColor(free) {
  if (free >= 6) {
    return '#7BC96F';
  }

  if (free >= 3) {
    return '#F4B942';
  }

  return '#E76F51';
}

export default function App() {
  const [screen, setScreen] = useState('home');

  const [selectedBench, setSelectedBench] = useState(null);
  const [selectedSeat, setSelectedSeat] = useState(null);

  const [amount, setAmount] = useState('');
  const [paymentError, setPaymentError] = useState('');

  const [selectedPayment, setSelectedPayment] = useState(null);

  const [totalPaid, setTotalPaid] = useState(37);
  const [sitCount, setSitCount] = useState(6);

  const [activeSeat, setActiveSeat] = useState(null);
  const [leaving, setLeaving] = useState(false);
  const [countdown, setCountdown] = useState(10);

  const [voted, setVoted] = useState(false);
  const [selectedVote, setSelectedVote] = useState(null);

  const [poolAmount, setPoolAmount] = useState(581150);

  const [customAmount, setCustomAmount] = useState('');

  useEffect(() => {
    if (!leaving) {
      return;
    }

    if (countdown <= 0) {
      setLeaving(false);
      setCountdown(10);
      setActiveSeat(null);
      setSelectedSeat(null);
      setScreen('home');
      return;
    }

    const timer = setTimeout(() => {
      setCountdown((value) => value - 1);
    }, 1000);

    return () => clearTimeout(timer);
  }, [leaving, countdown]);

  const openBench = (bench) => {
    setSelectedBench(bench);
    setSelectedSeat(null);
    setScreen('bench');
  };

  const seats = selectedBench
    ? Array.from({ length: 10 }, (_, index) => ({
        id: index + 1,
        occupied:
          index >= selectedBench.free ||
          (
            activeSeat &&
            activeSeat.benchId === selectedBench.id &&
            activeSeat.seatId === index + 1
          ),
      }))
    : [];

  const handleAmountChange = (text) => {
    let cleaned = text.replace(/[^0-9,.]/g, '');

    cleaned = cleaned.replace('.', ',');

    const parts = cleaned.split(',');

    if (parts.length > 2) {
      cleaned =
        parts[0] +
        ',' +
        parts.slice(1).join('');
    }

    if (parts.length === 2) {
      cleaned =
        parts[0] +
        ',' +
        parts[1].slice(0, 2);
    }

    setCustomAmount(cleaned);
    setPaymentError('');
  };

  const choosePayment = (value) => {
    setSelectedPayment(value);
    setCustomAmount('');
    setPaymentError('');
  };

  const getFinalAmount = () => {
    if (selectedPayment === 'custom') {
      return parseFloat(
        customAmount.replace(',', '.')
      );
    }

    return selectedPayment;
  };

  const handlePayment = () => {
    const finalAmount = getFinalAmount();

    if (!finalAmount || isNaN(finalAmount)) {
      setPaymentError('Wybierz kwotę płatności.');
      return;
    }

    if (finalAmount < 1) {
      setPaymentError(
        'Kwota poniżej minimalnej płatności 1 zł.'
      );
      return;
    }

    setPaymentError('');

    setTotalPaid((value) => value + finalAmount);
    setPoolAmount((value) => value + finalAmount / 2);
    setSitCount((value) => value + 1);

    setActiveSeat({
      benchId: selectedBench.id,
      seatId: selectedSeat,
      amount: finalAmount,
    });

    setScreen('success');
  };

  const startLeaving = () => {
    setLeaving(true);
    setCountdown(10);
  };

  const cancelLeaving = () => {
    setLeaving(false);
    setCountdown(10);
  };

  const submitVote = () => {
    if (!selectedVote) {
      return;
    }

    setVoted(true);
  };

  /*
   * =========================
   * EKRAN AKTYWNEGO MIEJSCA
   * =========================
   */

  if (screen === 'active') {
    return (
      <View style={styles.centerScreen}>
        <Text style={styles.activeEmoji}>
          🪑
        </Text>

        <Text style={styles.successTitle}>
          MIEJSCE ZAJĘTE
        </Text>

        <Text style={styles.successText}>
          Ławka #{activeSeat?.benchId}
        </Text>

        <Text style={styles.bigSeat}>
          Miejsce {activeSeat?.seatId}
        </Text>

        <View style={styles.activeBox}>
          <Text style={styles.activeBoxTitle}>
            Ciesz się chwilą 💚
          </Text>

          <Text style={styles.activeBoxText}>
            Możesz siedzieć tak długo, jak chcesz.
          </Text>

          <Text style={styles.activeBoxText}>
            Po opuszczeniu miejsca wybierz
            „OPUŚCIŁAM MIEJSCE”.
          </Text>
        </View>

        {!leaving ? (
          <TouchableOpacity
            style={styles.leaveButton}
            onPress={startLeaving}
          >
            <Text style={styles.leaveButtonText}>
              OPUŚCIŁAM MIEJSCE
            </Text>
          </TouchableOpacity>
        ) : (
          <View style={styles.countdownBox}>
            <Text style={styles.countdownNumber}>
              {countdown}
            </Text>

            <Text style={styles.countdownText}>
              Ławka zamknie się za {countdown} sekund
            </Text>

            <TouchableOpacity
              style={styles.cancelButton}
              onPress={cancelLeaving}
            >
              <Text style={styles.cancelButtonText}>
                WRÓCIŁAM
              </Text>
            </TouchableOpacity>
          </View>
        )}
      </View>
    );
  }

  /*
   * =========================
   * EKRAN SUKCESU
   * =========================
   */

  if (screen === 'success') {
    return (
      <View style={styles.centerScreen}>
        <Text style={styles.successEmoji}>
          🪑❤️
        </Text>

        <Text style={styles.successTitle}>
          DZIĘKUJEMY!
        </Text>

        <Text style={styles.successText}>
          Miejsce {selectedSeat} jest teraz Twoje.
        </Text>

        <Text style={styles.successText}>
          Wpłacono{' '}
          {getFinalAmount()
            ?.toFixed(2)
            .replace('.', ',')}{' '}
          zł.
        </Text>

        <View style={styles.splitBox}>
          <Text style={styles.splitTitle}>
            Twoja wpłata
          </Text>

          <Text style={styles.splitText}>
            💚 50% → rozwój SIADAJ
          </Text>

          <Text style={styles.splitText}>
            ❤️ 50% → miesięczna pomoc
          </Text>
        </View>

        <TouchableOpacity
          style={styles.button}
          onPress={() => setScreen('active')}
        >
          <Text style={styles.buttonText}>
            USIĄDŹ 🪑
          </Text>
        </TouchableOpacity>
      </View>
    );
  }

  /*
   * =========================
   * PŁATNOŚĆ
   * =========================
   */

  if (screen === 'payment') {
    return (
      <View style={styles.screen}>
        <ScrollView
          showsVerticalScrollIndicator={false}
          contentContainerStyle={styles.scrollContent}
        >
          <Text style={styles.topBack}
            onPress={() => setScreen('bench')}
          >
            ← WRÓĆ
          </Text>

          <Text style={styles.title}>
            Wybierz kwotę
          </Text>

          <Text style={styles.paymentInfo}>
            Miejsce {selectedSeat}
          </Text>

          <Text style={styles.minimumInfo}>
            Minimalna płatność: 1 zł
          </Text>

          <View style={styles.paymentGrid}>
            {[1, 2, 5, 10].map((value) => (
              <TouchableOpacity
                key={value}
                style={[
                  styles.paymentTile,
                  selectedPayment === value &&
                    styles.paymentTileSelected,
                ]}
                onPress={() => choosePayment(value)}
              >
                <Text
                  style={[
                    styles.paymentTileNumber,
                    selectedPayment === value &&
                      styles.paymentTileNumberSelected,
                  ]}
                >
                  {value} zł
                </Text>
              </TouchableOpacity>
            ))}
          </View>

          <TouchableOpacity
            style={[
              styles.customTile,
              selectedPayment === 'custom' &&
                styles.paymentTileSelected,
            ]}
            onPress={() => choosePayment('custom')}
          >
            <Text style={styles.customTileText}>
              ✏️ WŁASNA KWOTA
            </Text>
          </TouchableOpacity>

          {selectedPayment === 'custom' && (
            <View style={styles.amountContainer}>
              <TextInput
                style={styles.amountInput}
                value={customAmount}
                onChangeText={handleAmountChange}
                placeholder="np. 7,50"
                placeholderTextColor="#999999"
                keyboardType="decimal-pad"
                maxLength={10}
                autoFocus
              />

              <Text style={styles.currency}>
                zł
              </Text>
            </View>
          )}

          {selectedPayment && (
            <View style={styles.chosenAmountBox}>
              <Text style={styles.chosenAmountLabel}>
                Wybrana kwota
              </Text>

              <Text style={styles.chosenAmount}>
                {selectedPayment === 'custom'
                  ? customAmount || '0,00'
                  : selectedPayment.toFixed(2).replace('.', ',')}{' '}
                zł
              </Text>
            </View>
          )}

          {paymentError !== '' && (
            <View style={styles.errorBox}>
              <Text style={styles.errorText}>
                ⚠️ {paymentError}
              </Text>
            </View>
          )}

          <View style={styles.smallInfoBox}>
            <Text style={styles.smallInfo}>
              💚 50% wpłaty wspiera rozwój SIADAJ
              {'\n'}
              ❤️ 50% trafia do miesięcznej puli pomocy
            </Text>
          </View>

          <TouchableOpacity
            style={[
              styles.payButton,
              !selectedPayment &&
                styles.disabledButton,
            ]}
            onPress={handlePayment}
          >
            <Text style={styles.payButtonText}>
              ZAPŁAĆ
            </Text>
          </TouchableOpacity>
        </ScrollView>
      </View>
    );
  }

  /*
   * =========================
   * ŁAWKA
   * =========================
   */

  if (screen === 'bench' && selectedBench) {
    return (
      <View style={styles.screen}>
        <ScrollView
          showsVerticalScrollIndicator={false}
          contentContainerStyle={styles.scrollContent}
        >
          <TouchableOpacity
            onPress={() => setScreen('map')}
          >
            <Text style={styles.topBack}>
              ← MAPA
            </Text>
          </TouchableOpacity>

          <Text style={styles.title}>
            🪑 {selectedBench.name}
          </Text>

          <Text style={styles.availability}>
            <Text style={styles.greenText}>
              {selectedBench.free} wolnych
            </Text>
            {'  •  '}
            <Text style={styles.redText}>
              {10 - selectedBench.free} zajętych
            </Text>
          </Text>

          <Text style={styles.chooseText}>
            Wybierz miejsce
          </Text>

          <View style={styles.seatsContainer}>
            {seats.map((seat) => (
              <TouchableOpacity
                key={seat.id}
                disabled={seat.occupied}
                onPress={() =>
                  setSelectedSeat(seat.id)
                }
                style={[
                  styles.seat,
                  seat.occupied &&
                    styles.occupiedSeat,
                  selectedSeat === seat.id &&
                    styles.selectedSeat,
                ]}
              >
                <Text style={styles.seatNumber}>
                  {seat.id}
                </Text>

                <Text style={styles.seatStatus}>
                  {seat.occupied
                    ? 'ZAJĘTE'
                    : 'WOLNE'}
                </Text>
              </TouchableOpacity>
            ))}
          </View>

          {selectedSeat && (
            <View style={styles.selectionBox}>
              <Text style={styles.selectionText}>
                Miejsce {selectedSeat}
              </Text>

              <Text style={styles.priceText}>
                Wybierz kwotę, którą chcesz wpłacić
              </Text>

              <TouchableOpacity
                style={styles.payButton}
                onPress={() => {
                  setSelectedPayment(null);
                  setCustomAmount('');
                  setPaymentError('');
                  setScreen('payment');
                }}
              >
                <Text style={styles.payButtonText}>
                  PRZEJDŹ DO PŁATNOŚCI
                </Text>
              </TouchableOpacity>
            </View>
          )}
        </ScrollView>
      </View>
    );
  }

  /*
   * =========================
   * POMAGAMY
   * =========================
   */

  if (screen === 'help') {
    return (
      <View style={styles.screen}>
        <ScrollView
          showsVerticalScrollIndicator={false}
          contentContainerStyle={styles.scrollContent}
        >
          <Text style={styles.title}>
            ❤️ Pomagamy
          </Text>

          <Text style={styles.sectionSubtitle}>
            Twoje wpłaty zamieniają się w realną pomoc.
          </Text>

          <View style={styles.poolCard}>
            <Text style={styles.poolSmall}>
              AKTUALNA PULA POMOCY
            </Text>

            <Text style={styles.poolAmount}>
              {poolAmount.toLocaleString('pl-PL')}{' '}
              zł
            </Text>

            <Text style={styles.poolDescription}>
              Zebrane w tym miesiącu
            </Text>
          </View>

          <Text style={styles.sectionTitle}>
            🗳️ Głosowanie
          </Text>

          {!voted ? (
            <>
              <Text style={styles.voteDescription}>
                Wybierz, gdzie powinna trafić
                miesięczna pula.
              </Text>

              {helpCategories.map((category) => (
                <TouchableOpacity
                  key={category.id}
                  onPress={() =>
                    setSelectedVote(category.id)
                  }
                  style={[
                    styles.voteCard,
                    selectedVote === category.id &&
                      styles.voteCardSelected,
                  ]}
                >
                  <Text style={styles.voteEmoji}>
                    {category.emoji}
                  </Text>

                  <View style={styles.voteContent}>
                    <Text style={styles.voteName}>
                      {category.name}
                    </Text>

                    <Text style={styles.votePercentage}>
                      Obecnie {category.percentage}%
                    </Text>
                  </View>

                  <View
                    style={[
                      styles.radio,
                      selectedVote === category.id &&
                        styles.radioSelected,
                    ]}
                  />
                </TouchableOpacity>
              ))}

              <TouchableOpacity
                style={[
                  styles.voteButton,
                  !selectedVote &&
                    styles.disabledButton,
                ]}
                onPress={submitVote}
              >
                <Text style={styles.voteButtonText}>
                  ODDAJ GŁOS
                </Text>
              </TouchableOpacity>
            </>
          ) : (
            <View style={styles.votedBox}>
              <Text style={styles.votedEmoji}>
                ❤️
              </Text>

              <Text style={styles.votedTitle}>
                GŁOS ZAPISANY
              </Text>

              <Text style={styles.votedText}>
                Twój głos został dodany do
                miesięcznego głosowania.
              </Text>

              <Text style={styles.votedText}>
                Wyniki zostaną podsumowane pod
                koniec miesiąca.
              </Text>
            </View>
          )}

          <Text style={styles.sectionTitle}>
            📜 Historia pomocy
          </Text>

          {previousHelp.map((item) => (
            <View
              key={item.month}
              style={styles.historyCard}
            >
              <Text style={styles.historyEmoji}>
                {item.emoji}
              </Text>

              <View style={styles.historyContent}>
                <Text style={styles.historyMonth}>
                  {item.month}
                </Text>

                <Text style={styles.historyCause}>
                  Cel: {item.cause}
                </Text>

                <Text style={styles.historyAmount}>
                  Przekazano: {item.amount}
                </Text>
              </View>
            </View>
          ))}

          <View style={styles.transparencyBox}>
            <Text style={styles.transparencyTitle}>
              🔎 Pełna przejrzystość
            </Text>

            <Text style={styles.transparencyText}>
              Po zakończeniu każdego miesiąca
              pokazujemy zebrany wynik, zwycięski
              cel oraz przekazaną kwotę.
            </Text>

            <Text style={styles.transparencyText}>
              Transfer środków następuje około
              10. dnia kolejnego miesiąca.
            </Text>
          </View>
        </ScrollView>
      </View>
    );
  }

  /*
   * =========================
   * PROFIL
   * =========================
   */

  if (screen === 'profile') {
    return (
      <View style={styles.screen}>
        <ScrollView
          showsVerticalScrollIndicator={false}
          contentContainerStyle={styles.scrollContent}
        >
          <Text style={styles.profileIcon}>
            👤
          </Text>

          <Text style={styles.title}>
            Twój profil
          </Text>

          <Text style={styles.profileSubtitle}>
            Użytkownik SIADAJ
          </Text>

          <View style={styles.statsGrid}>
            <View style={styles.statCard}>
              <Text style={styles.statNumber}>
                {sitCount}
              </Text>

              <Text style={styles.statLabel}>
                skorzystań
              </Text>
            </View>

            <View style={styles.statCard}>
              <Text style={styles.statNumber}>
                {totalPaid.toFixed(2).replace('.', ',')} zł
              </Text>

              <Text style={styles.statLabel}>
                wpłat
              </Text>
            </View>
          </View>

          <View style={styles.profileCard}>
            <Text style={styles.profileCardTitle}>
              ❤️ Twój wpływ
            </Text>

            <Text style={styles.profileCardText}>
              Dzięki Twoim wpłatom część środków
              trafia co miesiąc do wybranego celu
              społecznego.
            </Text>

            <View style={styles.profileImpact}>
              <Text style={styles.profileImpactNumber}>
                {(totalPaid / 2)
                  .toFixed(2)
                  .replace('.', ',')}{' '}
                zł
              </Text>

              <Text style={styles.profileImpactLabel}>
                przekazanych na pomoc
              </Text>
            </View>
          </View>

          <View style={styles.profileCard}>
            <Text style={styles.profileCardTitle}>
              🪑 Ostatnia aktywność
            </Text>

            <View style={styles.activityRow}>
              <Text style={styles.activityEmoji}>
                🪑
              </Text>

              <View>
                <Text style={styles.activityTitle}>
                  Skorzystanie z SIADAJ
                </Text>

                <Text style={styles.activityDate}>
                  Ostatnia wpłata
                </Text>
              </View>

              <Text style={styles.activityAmount}>
                - 5,00 zł
              </Text>
            </View>

            <View style={styles.activityRow}>
              <Text style={styles.activityEmoji}>
                ❤️
              </Text>

              <View>
                <Text style={styles.activityTitle}>
                  Wsparcie społeczne
                </Text>

                <Text style={styles.activityDate}>
                  50% każdej wpłaty
                </Text>
              </View>

              <Text style={styles.activityAmount}>
                + pomoc
              </Text>
            </View>
          </View>

          <View style={styles.profileCard}>
            <Text style={styles.profileCardTitle}>
              🏆 Mały fakt
            </Text>

            <Text style={styles.profileCardText}>
              Każde użycie SIADAJ to nie tylko
              chwila odpoczynku. To także mała
              cegiełka w większej sprawie.
            </Text>
          </View>
        </ScrollView>
      </View>
    );
  }

  /*
   * =========================
   * MAPA
   * =========================
   */

  if (screen === 'map') {
    return (
      <View style={styles.screen}>
        <Text style={styles.mapTitle}>
          📍 Znajdź ławkę
        </Text>

        <Text style={styles.mapSubtitle}>
          Kliknij ławkę, aby zobaczyć dostępne miejsca
        </Text>

        <View style={styles.map}>
          <View style={styles.roadHorizontal} />
          <View style={styles.roadVertical} />

          {benches.map((bench) => (
            <TouchableOpacity
              key={bench.id}
              onPress={() => openBench(bench)}
              style={[
                styles.mapBench,
                {
                  backgroundColor:
                    getBenchColor(bench.free),
                  left: `${bench.x}%`,
                  top: `${bench.y}%`,
                },
              ]}
            >
              <Text style={styles.mapBenchText}>
                🪑
              </Text>

              <Text style={styles.mapBenchNumber}>
                {bench.free}
              </Text>
            </TouchableOpacity>
          ))}

          <View style={styles.userLocation}>
            <Text style={styles.locationText}>
              ●
            </Text>
          </View>
        </View>

        <View style={styles.legend}>
          <View style={styles.legendItem}>
            <View
              style={[
                styles.legendDot,
                { backgroundColor: '#7BC96F' },
              ]}
            />
            <Text>6–10 wolnych</Text>
          </View>

          <View style={styles.legendItem}>
            <View
              style={[
                styles.legendDot,
                { backgroundColor: '#F4B942' },
              ]}
            />
            <Text>3–5 wolnych</Text>
          </View>

          <View style={styles.legendItem}>
            <View
              style={[
                styles.legendDot,
                { backgroundColor: '#E76F51' },
              ]}
            />
            <Text>0–2 wolne</Text>
          </View>
        </View>

        <TouchableOpacity
          style={styles.backButton}
          onPress={() => setScreen('home')}
        >
          <Text style={styles.backButtonText}>
            ← WRÓĆ
          </Text>
        </TouchableOpacity>
      </View>
    );
  }

  /*
   * =========================
   * EKRAN GŁÓWNY
   * =========================
   */

  return (
    <View style={styles.container}>
      <Text style={[styles.cloud, styles.cloud1]}>
        ☁️
      </Text>

      <Text style={[styles.cloud, styles.cloud2]}>
        ☁️
      </Text>

      <Text style={[styles.cloud, styles.cloud3]}>
        ☁️
      </Text>

      <ScrollView
        showsVerticalScrollIndicator={false}
        contentContainerStyle={styles.homeContent}
      >
        <Text style={styles.logo}>
          SIADAJ
        </Text>

        <Text style={styles.subtitle}>
          Usiądź. Odpocznij. Pomóż.
        </Text>

        <TouchableOpacity
          style={styles.mainButton}
          onPress={() => setScreen('map')}
        >
          <Text style={styles.mainButtonText}>
            📍 ZNAJDŹ MNIE
          </Text>
